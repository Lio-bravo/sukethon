cmake_minimum_required(VERSION 3.10)
project(sukethon_gantry)
find_package(gazebo REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs)
include_directories(${GAZEBO_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
set(CMAKE_CXX_STANDARD 14)
add_library(sukethon_gantry SHARED gantry_plugin.cpp)
target_link_libraries(sukethon_gantry ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
